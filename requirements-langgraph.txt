# Dependencies of the LangGraph service (agenticRAG/langgraph_agents).
#
# THIS FILE IS THE WHOLE LIST. It used to be a partial one — five packages the
# service imports were missing, and it ran anyway only because
# agentic_rag_gemini/requirements.txt had been installed into the same
# environment. Anyone installing from this file alone, or building a container
# from it, got a backend that died on import.
#
# `tests/langgraph_agents/test_requirements_complete.py` now compares this file
# against the actual imports on every test run, so the gap cannot reopen
# silently.

# Core framework
langgraph>=0.2.0
langchain-core>=0.3.0

# LLM provider — DeepSeek via OpenAI-compatible
langchain-openai>=1.0.0,<2.0.0

# LLM fallback — Gemini. Imported lazily (inside the fallback functions), so a
# missing install shows up as a failed fallback at 3am rather than at start-up.
# Upper bound is deliberate: llm.py documents that response content shape is
# version-dependent (4.2.3 returns content blocks for some models where the code
# assumes a plain string). Re-verify before crossing a major.
langchain-google-genai>=4.2.3,<5.0.0

# MCP Integration
langchain-mcp-adapters>=0.1.0
mcp>=1.0.0

# Database
asyncpg>=0.29.0
pgvector>=0.3.0
# Reads the connection string from an SSM SecureString when VVA_PG_DSN_PARAM is
# set — the deployed path, where a DSN in a Lambda environment variable would sit
# in plaintext in the CloudFormation template. Declared here even though the
# Lambda runtime already ships it, because db/postgres.py imports it and a local
# run against SSM would otherwise fail with an ImportError. It is deliberately
# NOT in infra/lambda/crud_api/requirements.txt: adding ~100 MB to the deployment
# zip to ship what the runtime already provides.
boto3>=1.34.0
alembic>=1.13.0
# Alembic pulls this in anyway, but `alembic/env.py` imports it directly, so it
# is a first-class dependency and is declared as one.
SQLAlchemy>=2.0

# Embeddings — intfloat/multilingual-e5-small, 384-dim, CPU.
# Both sides of the pipeline go through shared/embedding.py: it prepends the
# `query: ` / `passage: ` prefixes that e5 REQUIRES, and a mismatch there
# degrades recall silently rather than failing.
#
# Brings torch with it (~2 GB). On a CPU-only box install the CPU wheel first to
# avoid pulling the CUDA build:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
sentence-transformers>=3.0
numpy>=1.26

# Config files — config/langgraph.yaml, config/mcp_servers.yaml, alembic

# FastAPI stack — pin to avoid starlette compat breaks

starlette>=1.0,<2.0
uvicorn[standard]>=0.30,<1.0
sse-starlette>=3.4,<4.0

# HTTP client (for VieNeu-TTS REST calls)
httpx>=0.27.0

# Redis — STM + task_result persistence (DB 0)
redis>=5.0.0,<6.0.0

# .env loader
python-dotenv>=1.0.0

# Web search (MCP backend) — self-hosted SearXNG via httpx (no pip dep needed)

# Pydantic

# Testing

pytest-asyncio>=0.23.0

# JWT verification — Cognito RS256 (PART A auth integration)
PyJWT[crypto]>=2.8.0
